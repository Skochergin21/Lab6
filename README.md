## Задача 1. Напишите программу, которая проверяет является ли введенный год високосным.

## Код

```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <locale.h>

int main()
{
    int yeas;          // год который вводим
    int ostatok;       // остаток от деления на 4
    int visokos = 0;   // 1 - високосный, 0 - нет

    setlocale(LC_ALL, "Rus");

    // приглашение и ввод года
    printf("Введите год: ");
    if (scanf("%d", &yeas) != 1 || yeas <= 0)
    {
        printf("Ошибка! Год должен быть целым числом больше 0\n");
        return 1;
    }

    // пункт 1 - считаем остаток от деления на 4
    ostatok = yeas % 4;
    printf("Остаток от деления на 4: %d\n", ostatok);

    if (ostatok == 0)
        printf("Если считать только по 4: год %d високосный\n", yeas);
    else
        printf("Если считать только по 4: год %d не високосный\n", yeas);

    // пункт 2 - точное условие (кратен 4 но не кратен 100, или кратен 400)
    if ((yeas % 4 == 0 && yeas % 100 != 0) || yeas % 400 == 0)
        visokos = 1;

    if (visokos == 1)
        printf("По точному условию: год %d високосный\n", yeas);
    else
        printf("По точному условию: год %d не високосный\n", yeas);

    return 0;
}
```

## Результат

```
Введите год: 2018
Остаток от деления на 4: 2
Если считать только по 4: год 2018 не високосный
По точному условию: год 2018 не високосный

Введите год: 1900
Остаток от деления на 4: 0
Если считать только по 4: год 1900 високосный
По точному условию: год 1900 не високосный

Введите год: 2020
Остаток от деления на 4: 0
Если считать только по 4: год 2020 високосный
По точному условию: год 2020 високосный
```

## Задача 2. Напишите программу для вычислений значения функции F(x).

Вариант 14

## Код

```c
/*
 * Вариант 14
 *
 *          | -x^3 + 9,      при x <= 13
 *   F(x) = |
 *          | 3 / (x + 1),   при x > 13
 *
 * Контрольные примеры (посчитал руками):
 * первое условие x <= 13:
 *   x = -2  -> -(-8) + 9 = 8 + 9  = 17
 *   x = 0   -> 0 + 9              = 9
 *   x = 2   -> -8 + 9             = 1
 *   x = 13  -> -2197 + 9          = -2188   (граница, тут еще первая формула)
 *
 * второе условие x > 13:
 *   x = 13,5 -> 3 / 14,5           = 0,206897  (сразу после границы)
 *   x = 14   -> 3 / 15             = 0,2
 *   x = 29   -> 3 / 30             = 0,1
 *
 * На ноль делить будет только при x = -1, а вторая формула
 * работает только когда x > 13, так что деления на ноль не будет.
 */
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <locale.h>

int main()
{
    double x;

    setlocale(LC_ALL, "Rus");

    printf("Введите x: ");
    if (scanf("%lf", &x) != 1)
    {
        printf("Ошибка! Надо ввести число\n");
        return 1;
    }

    // вывод результата через тернарную операцию прямо в printf
    printf("F(%g) = %f, формула: %s\n",
        x,
        (x <= 13) ? (-x * x * x + 9) : (3 / (x + 1)),
        (x <= 13) ? "-x^3 + 9 (x <= 13)" : "3/(x+1) (x > 13)");

    return 0;
}
```

## Результат

```

Введите x: 0
F(0) = 9,000000, формула: -x^3 + 9 (x <= 13)

Введите x: 13
F(13) = -2188,000000, формула: -x^3 + 9 (x <= 13)

Введите x: 13,5
F(13,5) = 0,206897, формула: 3/(x+1) (x > 13)

Введите x: 14
F(14) = 0,200000, формула: 3/(x+1) (x > 13)

Введите x: 29
F(29) = 0,100000, формула: 3/(x+1) (x > 13)
```
